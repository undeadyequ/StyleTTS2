    loss_mono = 0

# ldm loss
en_z_sampled, noise = model.decoder.q_sample(en_z)
z_score = model.decoder(en_z_sampled, F0_real_z, real_norm_z, s_z)
loss_diff = model.decoder.loss(z_score, noise)

g_loss = loss_params.lambda_mel * loss_mel + \
    loss_params.lambda_adv * loss_diff + \
    loss_params.lambda_mono * loss_mono + \
    loss_params.lambda_s2s * loss_s2s
